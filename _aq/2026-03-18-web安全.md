# 1. http header 认证头

> WWW-Authenticate: Digest realm="1b8b7cbe16af8c2872f87242", domain="/", qop="auth", nonce="f499ae99b5e92b74:1b8b7cbe16af8c2872f87242:19cf792d0db:3", opaque="799d5", algorithm="MD5", stale="FALSE"

- Digest: 表示使用摘要认证。
- realm: 认证域，用于指示哪些资源需要认证。通常是一个字符串，由服务器生成，客户端在计算响应时需要包含此域。
- domain: 指定认证起作用的URI空间。这里为"/"，表示整个域。
- qop: 保护质量。可以是"auth"（认证）或"auth-int"（认证和完整性保护）。这里为"auth"，表示只进行认证。
- nonce: 服务器生成的随机数，每次认证请求都不同。客户端在计算响应时要使用这个nonce。
- opaque: 服务器生成的一个字符串，客户端在请求中需要原样返回。
- algorithm: 指定生成摘要的算法。这里是MD5。
- stale: 指示nonce是否过期。如果为TRUE，客户端可以用相同的用户名和密码重新尝试，而不用重新输入密码。这里为FALSE，表示nonce没有过期。

## 1.1 摘要认证的流程

- 客户端访问一个需要认证的资源，服务器返回401状态码，并在WWW-Authenticate头部中提供上述信息。
- 客户端收到后，会要求用户输入用户名和密码，然后计算一个response值。
- 客户端重新请求该资源，并在Authorization头部中包括计算得到的response和其他信息。
- 客户端计算response的步骤（根据RFC 2617）：
  - 计算HA1 = MD5(username:realm:password)
  - 计算HA2 = MD5(method:uri) （如果qop为"auth"，如果是"auth-int"则包括实体体）
  - 计算response = MD5(HA1:nonce:nonceCount:cnonce:qop:HA2) （这里nonceCount和cnonce是客户端生成的）
- 然后客户端在Authorization头部中发送
  ```
  Authorization: Digest username="...", realm="...", nonce="...", uri="...", qop=auth, nc=..., cnonce="...", response="...", opaque="..."
  ```

```javascript
// 客户端需要计算的参数
function calculateDigestResponse(username, password, method, uri) {
	// 从服务器接收的参数
	const realm = '1b8b7cbe16af8c2872f87242';
	const nonce = 'f499ae99b5e92b74:1b8b7cbe16af8c2872f87242:19cf792d0db:3';
	const opaque = '799d5';
	const qop = 'auth';
	const algorithm = 'MD5';

	// 客户端生成的参数
	const nc = '00000001'; // 请求计数器（十六进制）
	const cnonce = '0a4f113b'; // 客户端nonce

	// 1. 计算 HA1 = MD5(username:realm:password)
	const ha1 = md5(`${username}:${realm}:${password}`);

	// 2. 计算 HA2 = MD5(method:uri)
	const ha2 = md5(`${method}:${uri}`);

	// 3. 计算 response = MD5(HA1:nonce:nc:cnonce:qop:HA2)
	const response = md5(`${ha1}:${nonce}:${nc}:${cnonce}:${qop}:${ha2}`);

	return response;
}
```

## 1.2 完成代码

```javascript
const crypto = require('crypto');
const axios = require('axios');

class DigestAuth {
	constructor(username, password) {
		this.username = username;
		this.password = password;
	}

	md5(str) {
		return crypto.createHash('md5').update(str).digest('hex');
	}

	parseChallenge(wwwAuthenticate) {
		// 解析 WWW-Authenticate 头
		const regex = /(\w+)="?([^",]+)"?/g;
		const params = {};
		let match;

		while ((match = regex.exec(wwwAuthenticate)) !== null) {
			params[match[1]] = match[2];
		}

		return params;
	}

	generateResponse(params, method, uri) {
		const { realm, nonce, qop, opaque, algorithm } = params;

		// 客户端生成的值
		const nc = '00000001';
		const cnonce = crypto.randomBytes(8).toString('hex');

		// 计算 HA1
		const ha1 = this.md5(`${this.username}:${realm}:${this.password}`);

		// 计算 HA2
		const ha2 = this.md5(`${method}:${uri}`);

		// 计算 response
		const response = this.md5(`${ha1}:${nonce}:${nc}:${cnonce}:${qop}:${ha2}`);

		return {
			username: this.username,
			realm,
			nonce,
			uri,
			qop,
			nc,
			cnonce,
			response,
			opaque,
			algorithm,
		};
	}

	async request(url, method = 'GET') {
		try {
			// 第一次请求，会收到401
			const response = await axios.get(url, { validateStatus: false });

			if (response.status === 401) {
				const authHeader = response.headers['www-authenticate'];
				const params = this.parseChallenge(authHeader);

				// 生成认证头
				const authParams = this.generateResponse(params, method, url);

				// 构建 Authorization 头
				const authString = Object.entries(authParams)
					.map(([k, v]) => `${k}="${v}"`)
					.join(', ');

				// 重新发送带认证的请求
				return axios.get(url, {
					headers: {
						Authorization: `Digest ${authString}`,
					},
				});
			}

			return response;
		} catch (error) {
			console.error('Digest auth failed:', error);
			throw error;
		}
	}
}

// 使用示例
const client = new DigestAuth('user123', 'password123');
client
	.request('http://example.com/protected')
	.then((response) => console.log('Success:', response.data))
	.catch((err) => console.error('Error:', err));
```

```javascript
const express = require('express');
const crypto = require('crypto');

const app = express();

// 模拟用户数据库
const users = {
	user123: {
		password: 'password123',
		realm: '1b8b7cbe16af8c2872f87242',
	},
};

// 存储有效的nonce（实际应用中用Redis等）
const validNonces = new Set();

function generateNonce() {
	const timestamp = Date.now().toString(16);
	const random = crypto.randomBytes(8).toString('hex');
	const realm = '1b8b7cbe16af8c2872f87242';
	const ipHash = crypto
		.createHash('md5')
		.update('127.0.0.1')
		.digest('hex')
		.slice(0, 12);
	const counter = Math.floor(Math.random() * 1000);

	return `${timestamp}:${realm}:${ipHash}:${counter}`;
}

// 受保护的路由
app.get('/api/protected', (req, res) => {
	const authHeader = req.headers.authorization;

	if (!authHeader || !authHeader.startsWith('Digest ')) {
		// 返回401要求认证
		const nonce = generateNonce();
		validNonces.add(nonce);

		res.set(
			'WWW-Authenticate',
			`Digest realm="1b8b7cbe16af8c2872f87242", ` +
				`domain="/", ` +
				`qop="auth", ` +
				`nonce="${nonce}", ` +
				`opaque="799d5", ` +
				`algorithm="MD5", ` +
				`stale="FALSE"`,
		);
		return res.status(401).send('Authentication required');
	}

	// 解析认证头
	const authString = authHeader.substring(7);
	const authParams = {};
	const regex = /(\w+)="([^"]*)"/g;
	let match;

	while ((match = regex.exec(authString)) !== null) {
		authParams[match[1]] = match[2];
	}

	// 验证nonce
	if (!validNonces.has(authParams.nonce)) {
		res.set(
			'WWW-Authenticate',
			`Digest realm="1b8b7cbe16af8c2872f87242", stale="TRUE"`,
		);
		return res.status(401).send('Stale nonce');
	}

	// 验证用户
	const user = users[authParams.username];
	if (!user) {
		return res.status(401).send('Invalid credentials');
	}

	// 计算期望的response
	const ha1 = crypto
		.createHash('md5')
		.update(`${authParams.username}:${user.realm}:${user.password}`)
		.digest('hex');

	const ha2 = crypto
		.createHash('md5')
		.update(`${req.method}:${authParams.uri}`)
		.digest('hex');

	const expectedResponse = crypto
		.createHash('md5')
		.update(
			`${ha1}:${authParams.nonce}:${authParams.nc}:${authParams.cnonce}:${authParams.qop}:${ha2}`,
		)
		.digest('hex');

	// 比较response
	if (authParams.response === expectedResponse) {
		// 认证成功
		validNonces.delete(authParams.nonce); // 移除已使用的nonce
		return res.json({ message: 'Access granted', data: 'Protected data' });
	} else {
		return res.status(401).send('Invalid credentials');
	}
});

app.listen(3000, () => {
	console.log('Server running on port 3000');
});
```

# 2. HTTP认证框架

- Basic（基本认证）
- Digest（摘要认证）
- Bearer（承载令牌认证，例如OAuth 2.0）
  > Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...
  > OAuth 2.0的核心，令牌即凭证,依赖HTTPS传输安全,现代API、单点登录(SSO),有一下几种变种：JWT、Opaque Token
- HOBA（基于HTTP的认证，RFC 7486）
  HTTP Origin-Bound Authentication, RFC 7486

  > WWW-Authenticate: HOBA realm="api", challenge="..."
  > 基于数字签名，绑定到HTTP源
  > 特点：无状态、可防止重放攻击
  > 使用：较少见，主要用于特定安全场景

- Mutual（相互认证）
  > 原理：客户端和服务器相互验证证书
  > 安全性：非常高
  > 场景：银行、政府、企业内网API
  > 实现：mTLS (相互TLS)
- Negotiate（协商认证，例如Kerberos）

  > WWW-Authenticate: Negotiate
  > WWW-Authenticate: NTLM

  > 原理：协商使用SPNEGO、Kerberos或NTLM
  > 场景：Windows域环境、企业内部系统
  > 协议：Kerberos、NTLM

- AWS4-HMAC-SHA256（AWS签名版本4）

  > Authorization: AWS4-HMAC-SHA256
  > Credential=AKIAIOSFODNN7EXAMPLE/20220101/us-east-1/s3/aws4_request,
  > SignedHeaders=host;x-amz-date,
  > Signature=fe5f80f77d5fa3beca038a248ff027d0445342fe2855ddc963176630326f1024

  > 原理：基于HMAC的请求签名
  > 场景：AWS API、云服务API
  > 变种：Azure Shared Key、GCP签名
