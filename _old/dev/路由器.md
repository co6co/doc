# 命令
```
mtd write /tmp/xxxx.bin OS1 -r #程序刷入OS1
sysupgrade -v /tmp/xxxx.bin    #使用xxx.bin升级系统，和mtd命令一样


dd if=/dev/mtd3 of=/tmp/eeprom.bin	#备份eeprom
mtd write breed.bin Bootloader #刷入breed

falsh erase 0x140000 0x600000	#擦除 flash 原有数据
								#0x140000  firmware 的地址，可以在 OpenWrt 官方的设备详情页面查看 openwrt.log 获取，
								#0x600000 表示设定空间大小，这个空间大小必须大于固件大小 0x5c0133
flash write 0x140000 0x80000000 0x600000	#写入固件
											#将保存在 0x80000000 位置大小为大小为0x600000 的固件写入0x0 to 0x140000的空间中。
boot flash 0x140000 #重启路由器， 在0x140000 处启动固件
								
```
