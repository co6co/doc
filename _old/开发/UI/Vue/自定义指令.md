# 版本1.x
```
Vue.directive('my-directive', {
    inserted(el,binding){
        document.title=el.dataset.title;
    },
    bind: function () {
        alert("bind");
      // 准备工作
      // 例如，添加事件处理器或只需要运行一次的高耗任务
    },
    update: function (newValue, oldValue) {
        alert("update");
      // 值更新时的工作
      // 也会以初始值为参数调用一次
    },
    unbind: function () {
        alert("unbind");
      // 清理工作
      // 例如，删除 bind() 添加的事件监听器
    }
  })
```
