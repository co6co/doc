# 0.基础知识
Windows是`分层`设计的操作系统

`用户态`: 只能受限的访问内存, 且不允许访问外围设备. 占用CPU的能力被剥夺, CPU资源可以被其他程序获取。
`内核态`: CPU可以访问内存所有数据, 包括外围设备, 如硬盘, 网卡. CPU也可以将自己从一个程序切换到另一个程序。

![windows系统](/static/系统/imgs/windows系统.jpg)

一个驱动程序就`对应一个硬件`。
举个例，当`系统总线`识别到有`硬件插入`时，CPU通过`中断`来通知操作系统，此时`操作系统`去寻找硬件设备对应的`驱动程序`（也可以说是`模块`），
最后完成`驱动程序`与`硬件设备`的`绑定`。

驱动程序可分为：
①即`插即用`驱动程序--------基本都是由硬件厂商完成编写
②`内核扩展`驱动程序--------与硬件无关，只是扩展内核的功能
③`文件系统`驱动程序---------是接收针对文件的I/O请求，在进一步将这些请求转变成真正针对于存储设备或网络设备的I/O请求

# 1. ssdt 系统服务描述符表
内核层它有`两张表`
`ssdt`（System Services Descriptor Table，系统服务描述符表）
	常用的系统服务
	
![ssdt](/static/系统/imgs/ssdt.png)
	
	每次调用一个应用层的API，都这样调用一个内核API
	
![api](/static/系统/imgs/api.png)
	
	比如：·`CreateFileA` ，我用od随便打开了一个exe,
		1. `CreateFileA`里面调用CreateFileW
		2. 进入 ·`CreateFileW`
		3. 开始调用ntdll中的NtCreateFile函数了
		```
		mov esi,dword ptr ds:[<&ntdll.NtCreateFile>]                                                             
		...
		...
		call esi
		```
		
		```
		#NtCreateFile 汇编
		776C55C8 ntdll.ZwC>  B8 42000000     mov eax,0x42
		776C55CD             BA 0003FE7F     mov edx,0x7FFE0300
		776C55D2             FF12            call dword ptr ds:[edx]
		776C55D4             C2 2C00         retn 0x2C
		```
		`0x42`也就是`SSDT表的数组索引`，通过这个索引找到内核函数,
		`0x7FFE0300`它对应的就是我上面图片应用层的通用函数（`ntdll.KiFastSystemCall`）。
		
		KiFastSystemCall的汇编代码：
		```
		776C70B0 ntdll.KiFastSystemCall          8BD4            mov edx,esp
		776C70B2                                 0F34            sysenter
		776C70B4 ntdll.KiFastSystemCallRet       C3              retn
		```
		4. 进入内核函数KiFastCallEntry
		```
		mov     esi, [ebx+_KPCR.PrcbData.CurrentThread];	#获取当前线程
		...
		...
		mov     edi, eax
		shr     edi, 8
		and     edi, 10h
		mov     ecx, edi
		add     edi, [esi+_KTHREAD.ServiceTable] ;			#就是获取是哪一张表
		...
		...
		mov     edx, [edi + _KSYSTEM_SERVICE_TABLE.ParamTableBase]
		```
首先是CreateFileA里面调用CreateFileW

# 2. 驱动
`IO管理器`实现了通用的框架，每个驱动程序的参数，返回值他们定义都是相同的,能够替换其他其他驱动程序

消息机制，消息会被封装成一个结构体：`MSG`。
做驱动开发的时候，消息被封装成了另外一个结构体，也就是`IRP(I/O Request Package)`

## 2.1 创建一个`设备对象`
在进行`应用程序`和`驱动程序`交互的时候`必须`要提供`一个设备对象`。 

创建好设备对象以后，使用`Winobj`查看,
可用 `windbg`工具查看结构体定义

## 2.2 `设置IRP`与`派遣函数`
派遣函数的设置
	在`_DRIVER_OBJECT`中`MajorFunction`成员，一般处理一些常用的请求，例如读写，创建，关闭等等。
	
## 2.3 使用
	应用层通过CreateFile来创建一个`符号链接`，完成应用层与内核层的通信

## 2.4 代码:

驱动代码：

```
#include <ntddk.h>

VOID UnloadDriver(PDRIVER_OBJECT pDriver)
{
	UNICODE_STRING usSymName;
	RtlInitUnicodeString(&usSymName, L"\\??\\LittleFish");
	if (pDriver->DeviceObject != NULL)
	{
		IoDeleteSymbolicLink(&usSymName);
		IoDeleteDevice(pDriver->DeviceObject);
		KdPrint(("删除设备成功\n"));
	}
}
//创建设备对象
NTSTATUS CreateDevice(PDRIVER_OBJECT pDriverObject)
{
	NTSTATUS		Status;
	PDEVICE_OBJECT	pDeviceObj;
	UNICODE_STRING	usDevName;
	UNICODE_STRING	usSymName;

	//初始化字符串
	RtlInitUnicodeString(&usDevName, L"\\Device\\LittleFish");

	Status = IoCreateDevice(pDriverObject, NULL, &usDevName, \
		FILE_DEVICE_UNKNOWN, FILE_DEVICE_SECURE_OPEN, \
		TRUE, &pDeviceObj);
	if (!NT_SUCCESS(Status))
	{
		return Status;
	}
	pDeviceObj->Flags |= DO_BUFFERED_IO;
	//创建符号链接
	RtlInitUnicodeString(&usSymName, L"\\??\\LittleFish");
	Status = IoCreateSymbolicLink(&usSymName, &usDevName);
	if (!NT_SUCCESS(Status))
	{
		IoDeleteDevice(pDeviceObj);
		return Status;
	}

	return STATUS_SUCCESS;
}
//派遣函数的固定写法
NTSTATUS CreateCompleteRoutine(PDEVICE_OBJECT pDeviceObject, PIRP pIrp)
{
	NTSTATUS Status;

	KdPrint(("创建\n"));
	Status = STATUS_SUCCESS;
	pIrp->IoStatus.Status = Status;
	pIrp->IoStatus.Information = 0;
	IoCompleteRequest(pIrp, IO_NO_INCREMENT);
	return STATUS_SUCCESS;
}
NTSTATUS CloseCompleteRoutine(PDEVICE_OBJECT pDeviceObject, PIRP pIrp)
{
	NTSTATUS Status;

	KdPrint(("关闭\n"));
	Status = STATUS_SUCCESS;
	pIrp->IoStatus.Status = Status;
	pIrp->IoStatus.Information = 0;

	IoCompleteRequest(pIrp, IO_NO_INCREMENT);
	return STATUS_SUCCESS;
}
NTSTATUS ReadCompleteRoutine(PDEVICE_OBJECT pDeviceObject, PIRP pIrp)
{
	NTSTATUS Status;

	KdPrint(("读\n"));
	Status = STATUS_SUCCESS;
	pIrp->IoStatus.Status = Status;
	pIrp->IoStatus.Information = 0;

	IoCompleteRequest(pIrp, IO_NO_INCREMENT);
	return STATUS_SUCCESS;
}
NTSTATUS WriteompleteRoutine(PDEVICE_OBJECT pDeviceObject, PIRP pIrp)
{
	NTSTATUS Status;

	KdPrint(("写\n"));
	Status = STATUS_SUCCESS;
	pIrp->IoStatus.Status = Status;
	pIrp->IoStatus.Information = 0;

	IoCompleteRequest(pIrp, IO_NO_INCREMENT);
	return STATUS_SUCCESS;
}
NTSTATUS DriverEntry(PDRIVER_OBJECT pDriver, PUNICODE_STRING pReg)
{
	NTSTATUS		Status;
	Status = CreateDevice(pDriver);
	if (!NT_SUCCESS(Status))
	{
		KdPrint(("创建设备失败\n"));
	}
	else
	{
		KdPrint(("创建设备成功\n"));
		KdPrint(("%p--%wZ\n", pDriver, pReg));
	}
	//派遣函数
	pDriver->MajorFunction[IRP_MJ_CREATE] = CreateCompleteRoutine;
	pDriver->MajorFunction[IRP_MJ_CLOSE] = CloseCompleteRoutine;
	pDriver->MajorFunction[IRP_MJ_READ] = ReadCompleteRoutine;
	pDriver->MajorFunction[IRP_MJ_WRITE] = WriteompleteRoutine;

	pDriver->DriverUnload = UnloadDriver;
	return STATUS_SUCCESS;
}
```

# 3. NT驱动
也叫`内核扩展`程序，编写 NT驱动 ， 要包含`ntddk.h`。
如果编写`WDM`或者`WDF`驱动，那么则要在代码中包含`特定的头文件`.

驱动加载过程：
	a.IO管理器加载驱动，
	b.加载以后为其分配内存，
	c.为驱动程序分配一个驱动对象和注册表配置信息，
	d.传递给驱动DriverEntry对象，
	e.驱动得到控制权。
```
#include <ntddk.h>

VOID UnloadDriver(PDRIVER_OBJECT pDriver)
{
	KdPrint(("UnloadDriver------------------\n"));
}
NTSTATUS DriverEntry(PDRIVER_OBJECT pDriver, PUNICODE_STRING pReg)
{
	pDriver->DriverUnload = UnloadDriver;
	KdPrint(("%p--%wZ\n", pDriver, pReg));
	return STATUS_SUCCESS;
}
```

- DriverEntry		驱动程序的入口 类似 main函数 固定的写法
	pDriver			驱动对象，IO管理器把控制权交给我们之前（创建驱动对象），然后把驱动对象交给主函数
	pReg			注册表信息
- UnloadDriver		卸载函数-是->回调函数，驱动程序卸载的时候的做一些清理工作
