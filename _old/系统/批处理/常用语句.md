# for
```
//Delims像刀子，用来切蛋糕，tokens像叉子--取蛋糕/块屑
# 注意 单引号
for /f "delims=" %f in ( 'dir /s/b *.*' ) do move "%f" D:\forder
//delims="、-." 1、饶天亮-玫瑰爱人.wma  ===>[1,饶天亮,玫瑰爱人,wma] 
//tokens=1,2  [1,饶天亮,玫瑰爱人,wma]  do echo %i,%j
for /f "tokens=1,3 delims=、-." %a in ( "1、饶天亮-玫瑰爱人.wma" ) do echo %a %b


```
"delims="  ==>取消默认的以`空格`作为分割
