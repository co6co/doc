# 1. 修改IP
```
//当前网络配置
esxcli network ip interface list
//设置IP
esxcli network ip set --interface=vmk0 --ipv4=192.168.1.10/24
//设置网关
esxcli network ip route ipv4 add --gateway=192.168.1.1
//重新启动VMkernel接口
esxcli network ip restart --interface=vmk0
```

# 2.  esxi主机配置管理网络选项是灰色的
- 启用 ESXi Shell
- ALT+F1键。这将带您进入命令行 shell 界面  
- 使用esxcli 命令重建标准交换机。

	```
	#列出您的 vmkernal 接口及配置
	esxcli network ip interface list
	#删除 vmk0（管理口）
	esxcli network ip interface remove -I vmk0
	
	# 标准端口组列表
	esxcli network vswitch
	# 从“vSwitch0”中删除了名为“'Management Network”的端口组
	esxcli network vswitch standard portgroup remove -p 'Management Network' -v vSwitch0
	#这从 vSwitch0 中删除了名为 vmnic0 的上行链路
	esxcli network vswitch standard uplink remove -u vmnic0 -v vSwitch0
	#这从 vSwitch0 中删除了名为 vmnic4 的上行链路
	esxcli network vswitch standard uplink remove -u vmnic4 -v vSwitch0 
	#vSwitch0内创建了一个名为 Management Network 的端口组
	esxcli network vswitch standard portgroup add -p 'Management Network' -v vSwitch0

	#建了一个名为 vmk0 的vmkernal 接口
	esxcli network ip interface add -i vmk0 -p 'Management Network' -m 1500
	# 将 ip 地址和子网掩码添加到 vmk0
	esxcli network ip interface ipv4 set -i vmk0 -I yourmanagementip -N subnetmak -t static
	esxcli network ip route ipv4 add -g gatewayip -n default
	# 将 vmnic0 作为上行链路添加到 vSwitch0
	esxcli network vswitch standard uplink add -u vmnic0 -v vSwitch0
	
	#重新启动主机的管理代理
	services.sh restart
	#ALT+F2 返回 DCUI 界面
	
	# 其他
	#将vSwitch0上的Service Console划分到vLan 10上，如果vLan号为0则不设置vLan
	esxcfg-vswitch -v 10 -p "Service Console" vSwitch0 
	#列出系统的网卡
	esxcfg-nics –l 
	#删除交换机
	#esxcfg-vswitch -D vSwitch1
	#删除交换机上的网卡
	esxcfg-vswitch -u vmnic1 vswitch2
	#删除portgroup
	esxcfg-vswitch -D portgroup vswitch1
	```
# 3. esxi 分布式交换机
	```
	# list Distributed Switch configuration
	esxcli network vswitch dvs vmware list 
	# 显示物理网卡及状态
	esxcli network nic list
	#
	# unlink/remove a vDS uplink
	esxcfg-vswitch -Q vmnic -V dvPort_ID_of_vmnic dvSwitch 
	esxcfg-vswitch -P vmnic -V unused_dvPort_ID dvSwitch # add a vDS uplink
	#删除现有 VMkernel 端口
	esxcli network ip interface remove --interface-name=vmkX

	```


# 4. 删除分布式交互机
	```
	# 列出所有交换机
	esxcfg-vswitch -l
	# 删除上连端口
	esxcfg-vswitch --del-dvp-uplink=vmnic1 --dvp=DVPORT_ID dvsName
	# 删除 分布式交换机
	net-dvs -d dvsName
	# Depending on the number of DvsPortset-N, you will need to run this additional command and replace the N with value observed from previous step.
	# 如果执行 esxcfg-vswitch -l  留下 DvsPortset-N 的结构可以，可以执行下面的命令，删除该结构
	vsish -e set /net/portsets/DvsPortset-0/destroy destroy
	#Note: If this command returns busy, it means there were still workloads or VMkernel interfaces using VDS that was not cleaned up as part of Step 0. You can ignore and move to next step

	# 删除vds数据库，[which is needed or else]重启后VDS重启后会重新创建。
	rm -f /etc/vmware/dvsdata.db
	# Finally, reboot the ESXi host and the VDS should now be completely removed. 
	```
