### 服务器端 - Python

首先确保安装了 `requests` 库：
```bash
pip install requests
```

然后创建一个 Python 文件，例如 `wechat_server.py`：
```python
import requests
import hashlib
import time
import random
from urllib.parse import quote_plus

def get_access_token(appid, secret):
    """
    获取 access_token
    """
    url = f"https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid={appid}&secret={secret}"
    response = requests.get(url)
    data = response.json()
    if 'access_token' in data:
        return data['access_token']
    else:
        raise Exception("Failed to fetch access token")

def create_jsapi_signature(token, timestamp, nonce_str, url):
    """
    生成微信 JS-SDK 签名
    """
    string1 = [token, timestamp, nonce_str, url]
    string1 = sorted(string1)
    string1 = ''.join(string1)
    hash_sha1 = hashlib.sha1()
    hash_sha1.update(string1.encode('utf-8'))
    return hash_sha1.hexdigest().upper()

def get_jsapi_ticket(access_token):
    """
    获取 jsapi_ticket
    """
    url = f"https://api.weixin.qq.com/cgi-bin/ticket/getticket?access_token={access_token}&type=jsapi"
    response = requests.get(url)
    data = response.json()
    if 'ticket' in data:
        return data['ticket']
    else:
        raise Exception("Failed to fetch jsapi_ticket")

def get_jsapi_config(appid, secret, url):
    """
    获取 JS-SDK 配置信息
    """
    access_token = get_access_token(appid, secret)
    jsapi_ticket = get_jsapi_ticket(access_token)
    
    timestamp = str(int(time.time()))
    nonce_str = ''.join(random.choice('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789') for _ in range(16))
    url = quote_plus(url)
    
    signature = create_jsapi_signature(jsapi_ticket, timestamp, nonce_str, url)
    
    config_info = {
        'appId': appid,
        'timestamp': timestamp,
        'nonceStr': nonce_str,
        'signature': signature,
    }
    return config_info

# 示例使用
if __name__ == '__main__':
    appid = 'your-app-id'
    secret = 'your-secret-key'
    url = 'http://example.com/page.html'

    config_info = get_jsapi_config(appid, secret, url)
    print(config_info)
```

### 客户端 - HTML

创建一个 HTML 文件，例如 `index.html`：
```html
<!DOCTYPE html>
<html lang="zh">
<head>
    <meta charset="UTF-8">
    <title>微信 JS-SDK 示例</title>
</head>
<body>
    <button id="share">分享到朋友圈</button>

    <script src="https://res.wx.qq.com/open/js/jweixin-1.6.0.js"></script>
    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // 这里假设 configInfo 已经从服务器端获取到了
            var configInfo = {
                appId: 'your-app-id',
                timestamp: 'your-timestamp',
                nonceStr: 'your-nonce-str',
                signature: 'your-signature',
            };

            wx.config({
                debug: false, // 开启调试模式
                appId: configInfo.appId,
                timestamp: configInfo.timestamp,
                nonceStr: configInfo.nonceStr,
                signature: configInfo.signature,
                jsApiList: [
                    'onMenuShareTimeline', // 分享到朋友圈
                    'onMenuShareAppMessage', // 分享给朋友
                ]
            });

            wx.ready(function () {
                // 在这里调用微信 JS API
                document.getElementById('share').addEventListener('click', function() {
                    wx.onMenuShareTimeline({
                        title: '这是一个测试分享', // 分享标题
                        link: 'http://example.com/page.html', // 分享链接
                        imgUrl: 'http://example.com/image.jpg', // 分享图标
                        success: function () {
                            alert('分享成功');
                        },
                        cancel: function () {
                            alert('取消分享');
                        }
                    });
                });
            });

            wx.error(function (res) {
                console.log(res);
            });
        });
    </script>
</body>
</html>
```

### 注意事项

- 替换 `your-app-id` 和 `your-secret-key` 为你的公众号的 `appid` 和 `secret`。
- 替换 `http://example.com/page.html` 为你实际的页面 URL。
- 确保你的 `appid` 和 `secret` 是正确的。
- 确保你的 `url` 参数是完整的 URL，包括协议头（例如 `http://` 或 `https://`）。

这个示例包含了从服务器端生成配置信息到客户端使用 JS-SDK 的完整流程。请确保按照微信的文档进行配置，并检查权限范围是否正确。如果遇到任何问题，请随时询问。
