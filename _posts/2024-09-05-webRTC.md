示例来梳理如何在服务器端和浏览器端实现从 WebRTC 到 RTMP 的流传输。我们将使用 `mediasoup` 作为服务器端的 WebRTC 处理库，并假设有一个 RTMP 推送插件或自定义逻辑来处理 RTMP 推送。

### 服务器端

#### 1. 安装依赖

首先安装必要的依赖：

```sh
npm install mediasoup mediasoup-client express body-parser
```

#### 2. 创建服务器端代码

创建一个简单的 Node.js 服务器来处理 mediasoup 的逻辑：

```javascript
const express = require('express');
const bodyParser = require('body-parser');
const mediasoup = require('mediasoup');
const { Device, DtlsSrtpCryptoSuite, DtlsIceCandidate } = require('mediasoup-client');

const app = express();
app.use(bodyParser.json());

let worker;
let router;
let rtpTransport;
let rtmpTransport;

// 创建 Worker 和 Router
async function createMediasoupComponents() {
  worker = new mediasoup.Worker();
  router = await worker.createRouter({ mediaCodecs: mediasoup.RTPCapabilities });

  // 创建 RTP Transport
  rtpTransport = await router.createWebRtcTransport({
    listenIps: ['0.0.0.0'],
    enabledRtpExtensions: [],
    srtp: {
      cryptoSuite: 'AES_CM_128_HMAC_SHA1_80',
      keyParams: 'aaee77ccbbffdd00112233aabbccddeeff'
    }
  });

  // 创建 RTMP Transport
  // 假设有 RTMP 推送插件或自定义逻辑
  rtmpTransport = await router.createRtpTransport({
    // RTMP Transport 的配置
  });
}

createMediasoupComponents();

// API for getting router RtpCapabilities
app.get('/router/rtpCapabilities', (req, res) => {
  res.json(router.rtpCapabilities);
});

// API for creating a Producer Transport
app.post('/router/transport/producer', async (req, res) => {
  const transport = await router.createWebRtcTransport({
    // 创建 Producer Transport 的参数
  });
  res.json({ transportId: transport.id });
});

// API for creating a Consumer Transport
app.post('/router/transport/consumer', async (req, res) => {
  const transport = await router.createWebRtcTransport({
    // 创建 Consumer Transport 的参数
  });
  res.json({ transportId: transport.id });
});

// API for producing a track
app.post('/router/transport/:transportId/produce', async (req, res) => {
  const { transportId } = req.params;
  const transport = await router.getWebRtcTransport(transportId);
  const producer = await transport.produce(req.body);
  res.json({ producerId: producer.id });
});

// API for consuming a track
app.post('/router/transport/:transportId/consume', async (req, res) => {
  const { transportId } = req.params;
  const transport = await router.getWebRtcTransport(transportId);
  const consumer = await transport.consume(req.body);
  res.json({ consumerId: consumer.id });
});

// Start the server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Server listening on port ${PORT}`));
```

### 浏览器端

#### 1. 创建 HTML 页面

创建一个简单的 HTML 页面来获取用户的媒体流：

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>WebRTC to RTMP</title>
</head>
<body>
  <video id="localVideo" autoplay playsinline></video>
  <button id="startButton">Start Stream</button>

  <script src="https://unpkg.com/mediasoup-client@latest/dist/mediasoup-client.min.js"></script>
  <script>
    const localVideo = document.getElementById('localVideo');
    const startButton = document.getElementById('startButton');

    navigator.mediaDevices.getUserMedia({ video: true, audio: true })
      .then(stream => {
        localVideo.srcObject = stream;
        return fetch('/router/rtpCapabilities')
          .then(res => res.json())
          .then(rtpCapabilities => {
            const device = new mediasoupClient.Device();
            device.load({ routerRtpCapabilities: rtpCapabilities });

            const producerTransport = new mediasoupClient.WebRtcTransport({
              id: 'producerTransport',
              iceParameters: { // 从服务器端获取
                usernameFragment: '',
                password: ''
              },
              iceCandidates: [], // 从服务器端获取
              dtlsParameters: { // 从服务器端获取
                role: 'auto',
                fingerprints: []
              },
              iceServers: [{ urls: 'stun:stun.l.google.com:19302' }]
            });

            stream.getTracks().forEach(track => {
              producerTransport.produce({ track }).then(producer => {
                // 发送 producer 信息给服务器端
                fetch('/router/transport/producer', {
                  method: 'POST',
                  headers: { 'Content-Type': 'application/json' },
                  body: JSON.stringify({ kind: track.kind, rtpParameters: producer.rtpParameters })
                }).then(res => res.json()).then(data => {
                  console.log('Producer created:', data);
                });
              });
            });

            startButton.addEventListener('click', () => {
              // 开始推流
              producerTransport.start();
            });
          });
      })
      .catch(err => console.error(err));
  </script>
</body>
</html>
```

#### 2. VUE
```
<template>
  <div>
    <video ref="localVideo" autoplay playsinline></video>
    <button @click="startStream">Start Stream</button>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue';
import { Device } from 'mediasoup-client';

const localVideo = ref(null);

const startStream = async () => {
  try {
    // 获取媒体流
    const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true });
    localVideo.value.srcObject = stream;

    // 加载 mediasoup-client 设备
    const device = new Device();
    await device.load({ routerRtpCapabilities: routerRtpCapabilities.value });

    // 创建 Producer Transport
    const producerTransport = new device.createProducerTransport({
      id: 'producerTransport',
      iceParameters: { usernameFragment: '', password: '' }, // 从服务器端获取
      iceCandidates: [], // 从服务器端获取
      dtlsParameters: { role: 'auto', fingerprints: [] }, // 从服务器端获取
      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }],
      appData: {}
    });

    // 当 ICE 候选者准备好时发送到服务器
    producerTransport.on('icecandidate', (event) => {
      if (event.candidate) {
        sendIceCandidate(event.candidate);
      }
    });

    // 产生 Producer 并发送到服务器
    stream.getTracks().forEach(async (track) => {
      const producer = await producerTransport.produce({
        track,
        appData: {}
      });
      sendProducer(producer);
    });

    // 启动 Transport
    await producerTransport.start();
  } catch (error) {
    console.error('Error starting stream:', error);
  }
};

// 发送 ICE 候选者到服务器
const sendIceCandidate = (candidate) => {
  fetch('/api/ice-candidate', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(candidate)
  })
  .then(response => response.json())
  .then(data => console.log('ICE candidate sent:', data))
  .catch(error => console.error('Failed to send ICE candidate:', error));
};

// 发送 Producer 信息到服务器
const sendProducer = (producer) => {
  fetch('/api/producer', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      id: producer.id,
      kind: producer.kind,
      rtpParameters: producer.rtpParameters
    })
  })
  .then(response => response.json())
  .then(data => console.log('Producer sent:', data))
  .catch(error => console.error('Failed to send producer:', error));
};

const routerRtpCapabilities = ref(null);

onMounted(async () => {
  // 获取服务器端的 RTP 能力
  const response = await fetch('/api/router-rtp-capabilities');
  const capabilities = await response.json();
  routerRtpCapabilities.value = capabilities;
});
</script>

<style scoped>
video {
  width: 100%;
  max-width: 640px;
}
</style>
```
#### 3. 与服务器端交互

在上述 HTML 页面中，我们通过 AJAX 请求与服务器端交互，传递 RTP 能力和 Producer 信息。服务器端接收到这些信息后，会创建相应的 Producer，并开始处理流。

### 注意事项

- 服务器端的 RTMP 推送逻辑需要自定义实现或使用社区提供的插件。
- 确保服务器端和客户端之间的通信安全，使用 HTTPS。
- 需要处理 ICE 交换和 DTLS 证书交换的过程。
- 确保服务器端有足够的带宽和处理能力来支持流媒体传输。

这个示例展示了如何从 WebRTC 获取媒体流并推送到服务器端，然后由服务器端负责将 RTP 转换为 RTMP 推送到 RTMP 服务器。实际应用中可能需要进一步优化和完善，特别是对于生产级别的应用来说。
