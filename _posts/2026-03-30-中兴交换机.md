# 1. S5860-28X

```
enable     # 特权模式
zxr10      # 特权密码
con t      # 配置模式 configure terminal
```

# 2. 端口

```
interface gei-0/1/1/1
switchport access vlan 100 # 有些交换机不支持在此模式下配置命令

arp
interface vlan100
    arp permanent 192.168.1.2 00ff.fa74.882e 100        # arp绑定mac地址
```

```
interface xgei-0/1/1/52
  ipv4-mixed-access-group ingress 1
  interface xgei-0/1/1/50 # 可以连续进入端口，不用exit


switchvlan-configuration        # vlan 配置模式
interface xgei-0/1/1/52
(config-swvlan-if-xgei-0/1/1/52)switchport mode  trunk
(config-swvlan-if-xgei-0/1/1/52)switchport trunk vlan 100
(config-swvlan-if-xgei-0/1/1/52) exit
```

# 3. 用户配置

```
enable secret level 15 XXXXXX  # 配置特权密码 有的设备 比如 S5860-28X 被厂家锁定设置了不生效【或者还需要配置什么才能生效】
aaa-authentication-template 2001  # 创建认证模板
[aaa-authen-templage]aaa-authentication-type local     # 认证类型为 本地
[aaa-authen-templage]exit
aaa-authorization-template 2001   #创建授权模板
[aaa-auther-templage]aaa-authorization-type local
[aaa-auther-templage]exit

#####  理解
#authentication-template 2001：能不能登录（认证）
#authorization-template 1 + bind 2001：登录后有什么权限（授权）
#####
authorization-template 1
bind aaa-authorization-template 2001
exit

system-user
strong-password check disable
authentication-template 1
role guest1 # 未验证
 taskgroup guest #system-admin = 管理员system-operator = 运维system-guest = 只读

[config-system-user-authen-temp]bind aaa-authentication-template 2001
[config-system-user-authen-temp]exit
authorization-template 1
[config-system-user-author-temp]bind aaa-authorization-template 2001
[config-system-user-auther-temp]exit
user-name guest
 password simple Guest@123
 role guest1
user-name admin
 bind authorization-template 1
 bind authentication-template 1
 password admin@123
 exi
```

# 4. 安全acl

```
ipv4-mixed-access-list 1 # 定义 名字为 1 的混合 acl
  rule 100 permit 54a0.5055.f9fb 0000.0000.0000 any any
  rule 1000 deny any any any
ipv4-mixed-access-group interface gei-0/1/1/1 ingress 1 # 在接口入方向调用该acl

# 也可直接在接口下调用
intface gei-0/1/1/1
  ipv4-mixed-access-group ingress 1

```
