# 1. 网络

```
ip addr # 查看IP得知
sudo ip link set ens33 up # 临时设置开启网卡

sudo nano /etc/netplan/00-installer-config.yaml # 永久开启网卡
# dhcp
network:
  version: 2
  ethernets:
    ens33:
      dhcp4: true
      optional: true
# static
network:
  version: 2
  ethernets:
    ens33:
      addresses:
        - 192.168.1.100/24
      routes:
        - to: default
          via: 192.168.1.1
      nameservers:
        addresses: [8.8.8.8, 1.1.1.1]
```

```
sudo netplan apply  # 应用配置
```

# 3. mysql

## 3.1 安装msql

```
# 下载官方仓库配置包
wget https://dev.mysql.com/get/mysql-apt-config_0.8.28-1_all.deb
# 安装该配置包，在弹窗中选择 `MySQL Server & Cluster` -> `mysql-5.7`
sudo dpkg -i mysql-apt-config_0.8.28-1_all.deb
# 如果被取消后将不会有弹出框 需要 重新配置 mysql-apt-config
#sudo dpkg-reconfigure mysql-apt-config

sudo apt update
sudo apt install -y mysql-server
```

## 3.1 卸载8.5

```

sudo apt remove --purge mysql-server mysql-client mysql-common -y
sudo apt autoremove -y
sudo apt autoclean

# 可选
sudo rm -rf /etc/mysql /var/lib/mysql
sudo rm -rf /var/log/mysql

```

## 3.2 无密码登录

```
sudo mysql -u root

>ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY '您的新密码';
>FLUSH PRIVILEGES;
>EXIT;
```

# 4 redis 安装和常用操作

配置文件为：`/etc/redis/redis.conf`,Redis 监听 `127.0.0.1:6379`

```
sudo apt update
sudo apt install redis-server # 安装
sudo systemctl start redis-server  # 启动
sudo systemctl enable redis-server # 设置为自动启动
redis-cli ping # 验证安装

sudo nano /etc/redis/redis.conf
#requirepass foobared #去掉前面的注释更换 foobared
# 重启服务
sudo systemctl restart redis-server
# 验证密码
redis-cli -a 你的密码
cc@u-server:~$ redis-cli -a xxxxxx
```
